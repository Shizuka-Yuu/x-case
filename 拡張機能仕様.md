### 2.2 エンゲージメント計算ロジック
```javascript
function calculateEngagementScore(data) {
  const metrics = data.metrics || {};
  const weights = {
    views: 0.1,
    likes: 0.3,
    replies: 0.4,
    reposts: 0.2,
    bookmarks: 0.1
  };
  
  return Object.entries(weights).reduce((score, [key, weight]) => {
    const value = metrics[key] || 0;
    const normalized = Math.log10(value + 1) / Math.log10(10000000); // 最大1000万で正規化
    return score + normalized * weight;
  }, 0);
}
```

## 3. レーダーチャート実装

### 3.1 データ構成
6つのデータ項目：
1. **複合スコア**: 論争比率とプロモーション度の複合指標
2. **表示数**: Views（対数スケール）
3. **いいね数**: Likes（対数スケール）
4. **リポスト数**: Reposts（対数スケール）
5. **返信数**: Replies（対数スケール）
6. **ブックマーク数**: Bookmarks（対数スケール）

### 3.2 複合スコア計算ロジック
論争比率とプロモーション度を複合化する計算式：
```javascript
function calculateCompositeScore(data) {
  const controversyRatio = getControversyRatio(data); // 0-2.0の範囲
  const promotionScore = getPromotionScore(data);   // 0-100の範囲
  
  // 論争比率を0-100に正規化
  const normalizedControversy = Math.min(controversyRatio * 50, 100);
  
  // 複合スコア計算（重み付け要検討）
  // 案1: 単純平均
  const compositeScore = (normalizedControversy + promotionScore) / 2;
  
  // 案2: 論争比率を重視（議論活性化の重要度を高く評価）
  // const compositeScore = (normalizedControversy * 0.6 + promotionScore * 0.4);
  
  // 案3: プロモーション度を重視（戦略的側面を高く評価）
  // const compositeScore = (normalizedControversy * 0.4 + promotionScore * 0.6);
  
  return compositeScore;
}
```

// case.jsに追加
function createRadarChart(data) {
  const ctx = document.getElementById('radarChart').getContext('2d');
  
  const metrics = data.metrics || {};
  const compositeScore = calculateCompositeScore(data);
  
  // 対数スケールで正規化
  const normalizeLog = (value) => {
    return Math.log10(value + 1) / Math.log10(10000000) * 100;
  };
  
  new Chart(ctx, {
    type: 'radar',
    data: {
      labels: ['複合スコア', '表示数', 'いいね数', 'リポスト数', '返信数', 'ブックマーク数'],
      datasets: [{
        label: 'エンゲージメント指標',
        data: [
          compositeScore,
          normalizeLog(metrics.views || 0),
          normalizeLog(metrics.likes || 0),
          normalizeLog(metrics.reposts || 0),
          normalizeLog(metrics.replies || 0),
          normalizeLog(metrics.bookmarks || 0)
        ],
        backgroundColor: 'rgba(102, 126, 234, 0.2)',
        borderColor: 'rgba(102, 126, 234, 1)',
        borderWidth: 2,
        pointBackgroundColor: 'rgba(102, 126, 234, 1)',
        pointBorderColor: '#fff',
        pointHoverBackgroundColor: '#fff',
        pointHoverBorderColor: 'rgba(102, 126, 234, 1)'
      }]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      scales: {
        r: {
          beginAtZero: true,
          max: 100,
          ticks: {
            stepSize: 20
          }
        }
      },
      plugins: {
        legend: {
          display: false
        },
        tooltip: {
          callbacks: {
            label: function(context) {
              const labels = ['複合スコア', '表示数', 'いいね数', 'リポスト数', '返信数', 'ブックマーク数'];
              const index = context.dataIndex;
              const rawValues = [
                compositeScore.toFixed(1) + '%',
                (metrics.views || 0).toLocaleString(),
                (metrics.likes || 0).toLocaleString(),
                (metrics.reposts || 0).toLocaleString(),
                (metrics.replies || 0).toLocaleString(),
                (metrics.bookmarks || 0).toLocaleString()
              ];
              return labels[index] + ': ' + rawValues[index];
            }
          }
        }
      }
    }
  });
}

// case.jsに追加
function createScatterChart() {
  const ctx = document.getElementById('scatterChart').getContext('2d');
  
  const chartData = archiveData.map(item => {
    const jsonData = loadJsonData(item.path.replace('.md', '.json'));
    const promotionScore = getPromotionScore(jsonData);
    const compositeScore = calculateCompositeScore(jsonData);
    const views = jsonData.metrics?.views || 0;
    
    return {
      x: promotionScore,                    // 志向性
      y: compositeScore,                   // 影響力
      r: Math.log10(views + 1) * 10,       // サイズ：リーチ
      color: getContentTypeColor(jsonData), // 色：コンテンツタイプ
      label: item.filename.replace('.md', ''),
      caseId: item.caseId
    };
  });
  
  new Chart(ctx, {
    type: 'bubble',
    data: {
      datasets: [{
        label: 'ケース分析',
        data: chartData,
        backgroundColor: chartData.map(d => d.color),
        borderColor: 'rgba(102, 126, 234, 1)',
        borderWidth: 1,
        pointRadius: 8,
        pointHoverRadius: 10
      }]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      scales: {
        x: {
          title: {
            display: true,
            text: '志向性 (%)'
          },
          min: 0,
          max: 100
        },
        y: {
          title: {
            display: true,
            text: '影響力 (%)'
          },
          min: 0,
          max: 100
        }
      },
      plugins: {
        tooltip: {
          callbacks: {
            label: function(context) {
              return context.raw.label + 
                     ' (志向性: ' + context.parsed.x + '%, ' +
                     '影響力: ' + context.parsed.y.toFixed(1) + '%, ' +
                     '表示数: ' + Math.pow(10, context.raw.r / 10 - 1).toLocaleString() + ')';
            }
          }
        }
      }
    }
  });
}

// 既存実装をそのまま継承（backup/case.jsより）
function updateBackgroundByPromotionScore(promotionScore) {
  const thumbnail = document.querySelector(".overview-thumbnail");
  if (!thumbnail) return;

  let backgroundImage = "";
  const config = window.CONFIG || {};

  if (promotionScore >= 0 && promotionScore <= 19) {
    backgroundImage = `url(${config.backgroundImages?.basePath || "img"}/${config.backgroundImages?.images.low || "low.webp"})`;
  } else if (promotionScore >= 20 && promotionScore <= 39) {
    backgroundImage = `url(${config.backgroundImages?.basePath || "img"}/${config.backgroundImages?.images.normal || "normal.webp"})`;
  } else if (promotionScore >= 40 && promotionScore <= 69) {
    backgroundImage = `url(${config.backgroundImages?.basePath || "img"}/${config.backgroundImages?.images.mid || "mid.webp"})`;
  } else if (promotionScore >= 70) {
    backgroundImage = `url(${config.backgroundImages?.basePath || "img"}/${config.backgroundImages?.images.high || "high.webp"})`;
  }

  // 背景画像を更新（現在の配置を維持）
  thumbnail.style.background = `white ${backgroundImage} no-repeat 50% 129%`;
  thumbnail.style.backgroundSize = "contain";
}

// 既存のupdateStructuralJudgmentScore関数内で呼び出し
function updateStructuralJudgmentScore(judgment) {
  const structuralScore = getPromotionScore(judgment);
  // 既存の背景画像更新処理を継承
  updateBackgroundByPromotionScore(structuralScore);
}
